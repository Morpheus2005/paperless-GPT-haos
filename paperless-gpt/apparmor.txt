# ============================================================================
# AppArmor profile: Paperless-GPT
# Includes the s6-overlay v3 rules required by the Home Assistant base image
# (see official docs: https://developers.home-assistant.io/blog/2022/05/12/s6-overlay-base-images)
# ============================================================================
#include <tunables/global>

profile paperless-gpt flags=(attach_disconnected) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  # --------------------------------------------------------------------------
  # S6-Overlay v3 (REQUIRED for HA base image — without these /init cannot exec)
  # --------------------------------------------------------------------------
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Network access for HTTP API calls to Paperless-NGX and LLM providers
  network inet,
  network inet6,

  # Allow access to the data volume
  /data/** rw,
  /data/ r,

  # Allow access to addon config
  /config/** r,
  /config/ r,

  # Allow access to share
  /share/** rw,
  /share/ r,

  # Allow running the paperless-gpt binary and its entrypoint
  /app/paperless-gpt rix,
  /app/entrypoint.sh rix,
  /app/** r,
  /app/ r,

  # Run script
  /run.sh rix,

  # Bashio
  /usr/bin/bashio rix,

  # System binaries
  /bin/** rix,
  /usr/bin/** rix,
  /usr/local/bin/** rix,

  # Temp files
  /tmp/** rw,
  /tmp/ rw,

  # /dev/urandom for crypto
  /dev/urandom r,
}
